cmake_minimum_required(VERSION 3.28)
project(WizardCard VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WIZARD_BUILD_CLIENT "Build SFML hot-seat client" ON)
include(CTest)
include(FetchContent)
function(wizard_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b EXCLUDE_FROM_ALL)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(wizard_core src/core/state.cpp src/core/engine.cpp src/core/view.cpp)
target_include_directories(wizard_core PUBLIC include)
add_library(wizard_content src/content/content.cpp src/content/replay.cpp)
target_link_libraries(wizard_content PUBLIC wizard_core nlohmann_json::nlohmann_json)
add_executable(wizard_cli src/cli/main.cpp)
target_link_libraries(wizard_cli PRIVATE wizard_content)
wizard_warnings(wizard_core)
wizard_warnings(wizard_content)
wizard_warnings(wizard_cli)
if(WIZARD_BUILD_CLIENT)
  set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 8835b6b955c96fa1ad1278a8dae6f05c4143effc EXCLUDE_FROM_ALL)
  FetchContent_MakeAvailable(SFML)
  add_library(wizard_runtime src/runtime/runtime.cpp)
  target_include_directories(wizard_runtime PUBLIC include)
  target_link_libraries(wizard_runtime PUBLIC SFML::Graphics SFML::Audio)
  add_executable(wizard_client src/client/main.cpp)
  target_link_libraries(wizard_client PRIVATE wizard_content wizard_runtime)
  wizard_warnings(wizard_runtime)
  wizard_warnings(wizard_client)
  install(TARGETS wizard_client RUNTIME DESTINATION .)
endif()
if(BUILD_TESTING)
  FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG 56809e5282f104c5c8b570e7c2996cdc352d94f1 EXCLUDE_FROM_ALL)
  FetchContent_MakeAvailable(Catch2)
  add_executable(wizard_tests tests/rules.cpp tests/content.cpp)
  target_link_libraries(wizard_tests PRIVATE wizard_content Catch2::Catch2WithMain)
  wizard_warnings(wizard_tests)
  target_compile_definitions(wizard_tests PRIVATE WIZARD_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
  include(Catch)
  catch_discover_tests(wizard_tests WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME content_validate COMMAND wizard_cli validate ${CMAKE_CURRENT_SOURCE_DIR}/assets)
  add_test(NAME scripted_match COMMAND wizard_cli smoke ${CMAKE_CURRENT_SOURCE_DIR}/assets ${CMAKE_CURRENT_BINARY_DIR}/smoke.json)
endif()
install(TARGETS wizard_cli RUNTIME DESTINATION .)
if(MSVC)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()
install(DIRECTORY assets/ DESTINATION assets)
install(FILES LICENSE README.md CHANGELOG.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY examples/ DESTINATION examples)
set(CPACK_PACKAGE_NAME WizardCard)
set(CPACK_PACKAGE_VERSION "0.4.0-dev")
set(CPACK_PACKAGE_FILE_NAME "WizardCard-0.4.0-dev-windows-x64")
set(CPACK_GENERATOR ZIP)
include(CPack)
