function(wizard_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

string(JSON WIZARD_CARD_COUNT LENGTH "${WIZARD_CATALOG}" definitions)
math(EXPR WIZARD_CARD_LAST "${WIZARD_CARD_COUNT} - 1")
foreach(WIZARD_CARD_INDEX RANGE 0 ${WIZARD_CARD_LAST})
  string(JSON WIZARD_CARD_FILE GET "${WIZARD_CATALOG}" definitions ${WIZARD_CARD_INDEX} file)
  get_filename_component(WIZARD_CARD_DIRECTORY "${WIZARD_CARD_FILE}" DIRECTORY)
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/assets/${WIZARD_CARD_FILE}"
    DESTINATION "assets/${WIZARD_CARD_DIRECTORY}")
endforeach()
